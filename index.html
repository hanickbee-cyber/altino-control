2. 파이썬 자율주행 코딩

명령어: go(좌, 우), steering(각도), stop(), await delay(초)

import js import asyncio def go(left, right): js.go(left, right) def steering(st): js.steering(st) def stop(): js.stop() # 학생들을 위한 직관적인 딜레이 함수 async def delay(sec): await asyncio.sleep(sec) # '실행하기' 버튼을 누르면 작동하는 함수 def execute_code(*args): # 1. textarea에서 학생이 쓴 텍스트 가져오기 raw_code = js.document.getElementById('student-code').value # 2. 백그라운드에서 async 함수 형태로 자동 감싸기 (들여쓰기 4칸 적용) indented_code = "\n".join([" " + line for line in raw_code.split("\n")]) wrapper_code = f"async def _student_main():\n{indented_code}" # 3. 사용할 함수들을 모아둔 가상 환경 만들기 local_env = { 'go': go, 'steering': steering, 'stop': stop, 'delay': delay, 'asyncio': asyncio } # 4. 문법 오류 체크 및 동적 실행 try: exec(wrapper_code, local_env, local_env) asyncio.ensure_future(local_env['_student_main']()) except Exception as e: # 오류 발생 시 브라우저 경고창으로 알려줌 js.alert(f"코드에 오류가 있습니다!\n\n{e}")
